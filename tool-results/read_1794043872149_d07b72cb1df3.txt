    66→/** Score per assessment over time. Clicking a point loads that review into the rubric above. */
    67→function TrendChart({points,selectedId,onSelect}:{points:{id:number;at:string;score:number}[];selectedId?:number|null;onSelect?:(id:number)=>void}){
    68→  if(points.length<2)return <p className="muted" style={{fontSize:12,textAlign:'center',padding:'24px 0'}}>Two or more reviews are needed to plot a trend.</p>;
    69→  const w=640,h=150,pad=22;
    70→  const step=(w-pad*2)/(points.length-1);
    71→  const y=(s:number)=>h-pad-s/100*(h-pad*2);
    72→  const path=points.map((p,i)=>`${i===0?'M':'L'}${(pad+i*step).toFixed(1)},${y(p.score).toFixed(1)}`).join(' ');
    73→  return (
    74→    <svg viewBox={`0 0 ${w} ${h}`} className="tr-trend" role="img" aria-label="Assessment score trend">
    75→      {[100,80,65,50].map(g=>(
    76→        <g key={g}>
    77→          <line x1={pad} x2={w-pad} y1={y(g)} y2={y(g)} stroke="var(--border)" strokeDasharray="3 5"/>
    78→          <text x={2} y={y(g)+3} style={{fontSize:10,fill:'var(--muted)'}}>{g}</text>
    79→        </g>
    80→      ))}
    81→      <line x1={pad} x2={w-pad} y1={y(65)} y2={y(65)} stroke="var(--red)" strokeOpacity=".45" strokeDasharray="4 3"/>
    82→      <path d={`${path} L${(pad+(points.length-1)*step).toFixed(1)},${h-pad} L${pad},${h-pad} Z`} fill="var(--accent)" opacity=".08"/>
    83→      <path d={path} fill="none" stroke="var(--accent)" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"/>
    84→      {points.map((p,i)=>{
    85→        const active=p.id===selectedId;
    86→        return (
    87→          <g key={p.id} onClick={()=>onSelect?.(p.id)} style={{cursor:onSelect?'pointer':undefined}}>
    88→            <circle cx={pad+i*step} cy={y(p.score)} r={active?5:3.5} fill={active?'var(--accent)':'var(--surface)'} stroke="var(--accent)" strokeWidth="2"/>

    90→            <text x={pad+i*step} y={h-5} textAnchor="middle" style={{fontSize:10,fill:'var(--muted)'}}>
    91→              {indiaDate(p.at,true)}

    93→          </g>

   112→/** Criterion attainment against the trainer's own average — the shape of a review at a glance. */
   113→function RadarChart({rows,benchmark,size=280}:{rows:RubricRow[];benchmark:number;size?:number}){
   114→  if(rows.length<3)return null;
   115→  const cx=size/2,cy=size/2,r=size*0.29,n=rows.length,gutter=52;
   116→  const angle=(i:number)=>i/n*Math.PI*2-Math.PI/2;
   117→  const pt=(i:number,frac:number)=>[cx+Math.cos(angle(i))*r*frac,cy+Math.sin(angle(i))*r*frac];
   118→  const poly=(frac:(i:number)=>number)=>rows.map((_,i)=>pt(i,Math.max(0,Math.min(1,frac(i)))).join(',')).join(' ');
   119→  // Truncated on length rather than on word count, so "Music and tempo" does not become "Music and".
   120→  const short=(s:string)=>s.length>18?s.slice(0,17).trimEnd()+'…':s;
   121→  // The viewBox is padded sideways so a long criterion label sitting on the left or right
   122→  // spoke is drawn inside the box rather than clipped by it.
   123→  return (
   124→    <svg viewBox={`${-gutter} 0 ${size+gutter*2} ${size}`} className="tr-radar" role="img" aria-label="Rubric attainment radar">
   125→      {[0.25,0.5,0.75,1].map(f=><polygon key={f} points={poly(()=>f)} fill="none" stroke="var(--border)"/>)}
   126→      {rows.map((_,i)=>{const[x,y]=pt(i,1);return <line key={i} x1={cx} y1={cy} x2={x} y2={y} stroke="var(--border)"/>;})}
   127→      <polygon points={poly(()=>benchmark/100)} fill="none" stroke="var(--muted)" strokeWidth="1.5" strokeDasharray="4 4"/>
   128→      <polygon points={poly(i=>rows[i].pct/100)} fill="var(--accent)" fillOpacity=".16" stroke="var(--accent)" strokeWidth="2"/>
   129→      {rows.map((row,i)=>{const[x,y]=pt(i,row.pct/100);return <circle key={row.category} cx={x} cy={y} r="3.5" fill={scoreColor(row.pct)} stroke="var(--surface)" strokeWidth="1.5"/>;})}
   130→      {rows.map((row,i)=>{const[x,y]=pt(i,1.3);return (
   131→        <text key={row.category} x={x} y={y} textAnchor="middle" dominantBaseline="middle" style={{fontSize:10,fontWeight:600,fill:'var(--muted)'}}>{short(row.category)}</text>
   132→      );})}
   133→    </svg>
   134→  );
   135→}

   261→export function TrainerReport({trainer,canManage=false,onRemoveReview}:{
   262→  trainer:Trainer;
   263→  /** Administrators only: shows the per-review remove control. The API enforces it too. */
   264→  canManage?:boolean;
   265→  /** Asked to delete one review. The page owns the confirmation and the refresh. */
   266→  onRemoveReview?:(review:Assessment)=>Promise<void>;
   267→}){
   268→  // Scoped to the trainer on screen for the same reason as `openOverride`: a selection made on
   269→  // one trainer's file must not carry into the next one.
   270→  const [selection,setSelection]=useState<{for:string;id:number}>();
   271→  const selectedId=selection?.for===trainer.name?selection.id:null;
   272→  const setSelectedId=(id:number)=>setSelection({for:trainer.name,id});
   273→  // Newest first, which is both the reading order of the history list and the order the
   274→  // "latest" card is taken from.
   275→  const newestFirst=useMemo(()=>[...trainer.assessments].sort((a,b)=>+new Date(b.submittedAt)-+new Date(a.submittedAt)),[trainer.assessments]);
   276→  // Open by default: the most recent review only. Everything older stays collapsed until
   277→  // somebody asks for it.
   278→  //
   279→  // Held as an override rather than as the state itself, so switching trainer does not need an
   280→  // effect to re-seed it: with no override, the open set is derived from this trainer's own
   281→  // latest review, which is the correct answer for whichever trainer is on screen.
   282→  const [openOverride,setOpenOverride]=useState<{for:string;ids:Set<number>}>();
   283→  const openIds=openOverride?.for===trainer.name
   284→    ? openOverride.ids
   285→    : new Set(newestFirst.slice(0,1).map(a=>a.id));
   286→  const setOpenIds=(ids:Set<number>)=>setOpenOverride({for:trainer.name,ids});
   287→  const [deletingId,setDeletingId]=useState<number|null>(null);
   288→  async function removeReview(review:Assessment){
   289→    if(!onRemoveReview)return;
   290→    setDeletingId(review.id);
   291→    try{await onRemoveReview(review);}finally{setDeletingId(null);}
   292→  }
   293→  const [rubricFilter,setRubricFilter]=useState<'all'|'strong'|'focus'>('all');
   294→  // Read on the client after mount: "last reviewed N days ago" must not be baked into the
   295→  // server-rendered markup, or the two disagree the moment the page is cached.
   296→  const [now,setNow]=useState(0);
   297→  useEffect(()=>{const id=setTimeout(()=>setNow(Date.now()),0);return()=>clearTimeout(id);},[]);
   298→
   299→  const toggleOpen=(id:number)=>{
   300→    const next=new Set(openIds);
   301→    if(next.has(id))next.delete(id);else next.add(id);
   302→    setOpenIds(next);
   303→  };
   304→
   305→  // Oldest first for the trend line; the history list below reverses it.
   306→  const sorted=useMemo(()=>[...trainer.assessments].sort((a,b)=>+new Date(a.submittedAt)-+new Date(b.submittedAt)),[trainer.assessments]);
   307→  const scored=useMemo(()=>sorted.filter(a=>a.score>0),[sorted]);
   308→  const avg=trainer.avgScore??0;
   309→  const latest=scored.at(-1)||sorted.at(-1)||null;
   310→  const delta=scored.length>1?scored.at(-1)!.score-scored.at(-2)!.score:0;
   311→  const best=scored.length?Math.max(...scored.map(a=>a.score)):0;
   312→  const worst=scored.length?Math.min(...scored.map(a=>a.score)):0;
   313→
   314→  const selected=useMemo(()=>trainer.assessments.find(a=>a.id===selectedId)||latest,[trainer.assessments,selectedId,latest]);
   315→  const selectedRubric=useMemo(()=>selected?.rubric??[],[selected]);
   316→  const filteredRubric=useMemo(()=>{
   317→    if(rubricFilter==='strong')return selectedRubric.filter(r=>r.pct>=80);
   318→    if(rubricFilter==='focus')return selectedRubric.filter(r=>r.pct<65);
   319→    return selectedRubric;
   320→  },[selectedRubric,rubricFilter]);
   321→
   322→  /** Averaged criterion attainment across every review — the durable pattern, not one class. */
   323→  const lifetimeRubric=useMemo(()=>{
   324→    const map=new Map<string,{score:number;weightage:number;n:number}>();
   325→    for(const a of trainer.assessments)for(const r of a.rubric){
   326→      const cur=map.get(r.category)??{score:0,weightage:0,n:0};
   327→      cur.score+=r.score;cur.weightage+=r.weightage;cur.n+=1;map.set(r.category,cur);
   328→    }
   329→    return [...map.entries()].map(([category,v])=>({
   330→      category,score:Math.round(v.score/v.n*10)/10,weightage:Math.round(v.weightage/v.n*10)/10,
   331→      pct:v.weightage>0?Math.round(v.score/v.weightage*100):0,
   332→    })).sort((a,b)=>a.pct-b.pct);
   333→  },[trainer.assessments]);
   334→
   335→  const evaluators=useMemo(()=>{
   336→    const map=new Map<string,{count:number;total:number}>();
   337→    for(const a of trainer.assessments){
   338→      const key=a.evaluator?.trim()||'Unattributed';
   339→      const cur=map.get(key)??{count:0,total:0};cur.count+=1;cur.total+=a.score;map.set(key,cur);
   340→    }
   341→    return [...map.entries()].map(([evaluator,v])=>({evaluator,count:v.count,avg:Math.round(v.total/v.count)})).sort((a,b)=>b.count-a.count);
   342→  },[trainer.assessments]);
   343→
   344→  const studios=useMemo(()=>{
   345→    const map=new Map<string,{count:number;total:number}>();
   346→    for(const a of trainer.assessments){
   347→      const key=a.studio?.split(',')[0].trim()||'Unspecified';
   348→      const cur=map.get(key)??{count:0,total:0};cur.count+=1;cur.total+=a.score;map.set(key,cur);
   349→    }
   350→    return [...map.entries()].map(([studio,v])=>({studio,count:v.count,avg:Math.round(v.total/v.count)})).sort((a,b)=>b.count-a.count);
   351→  },[trainer.assessments]);
   352→
   353→  const daysSince=latest&&now?Math.floor((now-+new Date(latest.submittedAt))/86400000):null;
   354→  const alerts:{tone:string;title:string;body:string}[]=[];
   355→  if(latest&&latest.score>0&&latest.score<65)alerts.push({tone:'red',title:'Coaching priority',body:`The latest weighted score is ${latest.score}%, below the 65% threshold.`});
   356→  if(delta<=-8)alerts.push({tone:'amber',title:`Down ${Math.abs(delta)} points`,body:'A regression since the previous assessment — review the weakest criteria.'});
   357→  if(delta>=8)alerts.push({tone:'green',title:`Up ${delta} points`,body:'A clear improvement since the last review.'});
   358→  if(trainer.issueCount>=3)alerts.push({tone:'red',title:`${trainer.issueCount} flagged concerns`,body:'Member feedback on this trainer is trending negative.'});
   359→  if(daysSince!==null&&daysSince>90)alerts.push({tone:'amber',title:'Review overdue',body:`Last assessed ${daysSince} days ago.`});
   360→  if(!trainer.assessments.length)alerts.push({tone:'amber',title:'Never assessed',body:'No evaluation is on file for this trainer yet.'});
   361→  if(trainer.complimentCount>=3&&trainer.issueCount===0)alerts.push({tone:'green',title:'Consistently praised',body:`${trainer.complimentCount} compliments, no concerns logged.`});
   362→
   363→  return (
   364→    <div className="tr-report">
