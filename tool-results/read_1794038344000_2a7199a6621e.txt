   395→// === METRIC CARD with flip, tooltip, sparkline, floating icon ===
   396→function MetricCard({
   397→  label, value, icon: Icon, trend, trendUp, tooltip, sparkData, color,
   398→}: {
   399→  label: string; value: number; icon: React.ComponentType<{ className?: string }>;
   400→  trend?: string; trendUp?: boolean; tooltip: string; sparkData: number[]; color: string;
   401→}) {
   402→  const [hovered, setHovered] = React.useState(false);
   403→  const max = Math.max(...sparkData, 1);
   404→  const sparkPoints = sparkData.map((v, i) => `${(i / (sparkData.length - 1)) * 100},${30 - (v / max) * 25}`).join(" ");
   405→
   406→  return (
   407→    <Tooltip>
   408→      <TooltipTrigger asChild>
   409→        <div
   410→          className="iris-card p-3 relative overflow-hidden cursor-pointer transition-all"
   411→          onMouseEnter={() => setHovered(true)}
   412→          onMouseLeave={() => setHovered(false)}
   413→          style={{ transform: hovered ? "translateY(-2px)" : undefined }}
   414→        >
   415→          {/* Floating background icon */}
   416→          <Icon className="absolute -top-1 -right-1 w-8 h-8 opacity-5" style={{ color, animation: hovered ? "metric-float 2s ease-in-out" : undefined }} />
   417→          {/* Sparkline */}
   418→          <svg className="absolute bottom-0 left-0 w-full h-8 opacity-30" viewBox="0 0 100 30" preserveAspectRatio="none">
   419→            <polyline points={sparkPoints} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   420→          </svg>
   421→          {/* Content */}
   422→          <div className="relative">
   423→            <div className="flex items-center justify-between mb-1">
   424→              <div className="text-[9px] font-medium text-muted-foreground uppercase tracking-wider">{label}</div>
   425→              <Icon className="w-3 h-3" style={{ color }} />
   426→            </div>
   427→            <div className="text-[18px] font-bold tabular-nums iris-ticker">{value}</div>
   428→            {trend && (
   429→              <div className={cn("text-[9px] font-medium mt-0.5 flex items-center gap-0.5", trendUp ? "text-emerald-500" : "text-rose-500")}>
   430→                {trendUp ? <TrendingUp className="w-2 h-2" /> : <TrendingDown className="w-2 h-2" />}
   431→                {trend}
   432→              </div>
   433→            )}
   434→          </div>
   435→          <style>{`@keyframes metric-float { 0%,100% { transform: translateY(0) rotate(0deg); } 50% { transform: translateY(-4px) rotate(8deg); } }`}</style>
   436→        </div>
   437→      </TooltipTrigger>
   438→      <TooltipContent side="bottom" className="text-[11px]">
   439→        {tooltip}
   440→      </TooltipContent>
   441→    </Tooltip>
   442→  );
   443→}

   446→function MarqueeStrip({ items, icon: Icon, color }: { items: string[]; icon: React.ComponentType<{ className?: string }>; color: string }) {
   447→  // Duplicate items for seamless loop
   448→  const doubled = [...items, ...items];

   451→      <div className="flex items-center gap-4 whitespace-nowrap" style={{ animation: "marquee-scroll 30s linear infinite" }}>
   452→        {doubled.map((item, i) => (
   453→          <React.Fragment key={i}>
   454→            <div className="flex items-center gap-1.5 text-[10px] font-medium text-muted-foreground">
   455→              <Icon className="w-2.5 h-2.5 shrink-0" style={{ color }} />
   456→              {item}
   457→            </div>
   458→            <span className="text-muted-foreground/20">|</span>
   459→          </React.Fragment>
   460→        ))}
   461→      </div>
   462→      <style>{`@keyframes marquee-scroll { from { transform: translateX(0); } to { transform: translateX(-50%); } }`}</style>
   463→    </div>
   464→  );
